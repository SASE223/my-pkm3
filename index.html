using UnityEngine; using System.Linq; #if UNITY_EDITOR using UnityEditor; #endif public class L_CraneController : MonoBehaviour { public enum LocalDepthAxis { X, Y, Z } [Header("深度方向（仅无区块时使用）")] public LocalDepthAxis depthLocalAxis = LocalDepthAxis.Z; public Transform shelf; public Transform containerAttachPoint; [Header("移动配置")] [Range(0.1f, 50f)] public float moveSpeed = 5f; public float decelerationDistance = 2f; public float arrivalThreshold = 0.05f; [Header("Mode1")] public float mode1EndVehicleRadius = 3f; public float mode1StartRayLength = 10f; public LayerMask mode1StartDetectionLayers = ~0; public bool mode1LoopEnabled; public int mode1LoopCount = 1; [Header("Mode2")] public float mode2StartCargoRadius = 3f; public float mode2EndVehicleRadius = 5f; public bool mode2LoopEnabled; public int mode2LoopCount = 1; [Header("重试设置")] public int maxPickupRetries = 1; [Header("调试")] public bool showGizmos = true; public Color rayPositiveColor = Color.green, rayNegativeColor = Color.cyan; public Color hitColor = Color.red, centerPointColor = Color.red; public Color mode1RayGizmoColor = Color.cyan; public float rayLength = 50f; public Vector3 rayOriginOffset = Vector3.zero; [Header("自动对齐")] public bool autoAlignStartPoint = true; private Vector3 worldDepthDir, worldPositiveDir, worldNegativeDir; private Vector2 shelfColumnDir2D; private int currentHeightLevel; private float targetCraneDepth, targetShelfColumnOffset, targetShelfHeightDrop; public CraneState currentState { get; private set; } = CraneState.Idle; public GameObject attachedContainer { get; private set; } public int currentLoopIndex { get; private set; } public bool IsMoving => isMoving; public bool IsRunning => isRunning; public bool HasAligned => hasAligned; private bool isMoving, hasAligned, isRunning; private Transform currentTarget; private Vehicle currentVehicle; private int pickupRetryCount = 0; private bool retryPickupFlag = false; private float targetVehicleDepth; private Vector3 cachedCenter; private MeshRenderer[] meshRenderers; private bool hasValidMesh; public bool hasPickedUpCargo, hasDroppedCargo; private float currentVerticalSpeed, currentMaxDrop, loweringStartZ; private bool isLowering; void Start() { meshRenderers = GetComponentsInChildren(true); int validCount = 0; foreach (var r in meshRenderers) if (r != null) { var mf = r.GetComponent(); if (mf && mf.sharedMesh) validCount++; } hasValidMesh = validCount > 0; if (useMeshCenter && !hasValidMesh) useMeshCenter = false; UpdateShelfColumnDirection(); UpdateCenterPoint(); AutoGetReferences(); ValidateReferences(); if (assignedZone) { worldDepthDir = assignedZone.DepthDirection; worldPositiveDir = worldDepthDir; worldNegativeDir = -worldDepthDir; Vector3 colWorld = assignedZone.ColumnDirectionWorld; Vector3 colLocal = transform.InverseTransformDirection(colWorld).normalized; shelfColumnDir2D = new Vector2(colLocal.x, colLocal.y).normalized; if (shelfColumnDir2D.sqrMagnitude < 0.01f) shelfColumnDir2D = Vector2.right; assignedZone.SetColumnDirection(transform.TransformDirection(new Vector3(shelfColumnDir2D.x, shelfColumnDir2D.y, 0))); if (autoAlignStartPoint) AutoAlignToZone(); Debug.Log($"[{gameObject.name}] 列方向同步完成，自动对齐：{autoAlignStartPoint}"); } else { UpdateDepthDirectionFromLocal(); } if (mode1Active) currentMode = CraneMode.Mode1; if (mode2Active) currentMode = CraneMode.Mode2; } /// 将龙门吊深度和本地起点对齐到区块矩阵列中心 void AutoAlignToZone() { if (!assignedZone || !crane) return; Vector3 columnCenterWorld = assignedZone.GetColumnCenterWorld(); // 移动龙门吊深度到列中心所在深度 float targetDepth = Vector3.Dot(columnCenterWorld, worldDepthDir); float currentDepth = Vector3.Dot(crane.position, worldDepthDir); crane.position += worldDepthDir * (targetDepth - currentDepth); // 将本地起点设为列中心在龙门吊本地空间的坐标（保持工作高度） Vector3 localCenter = crane.InverseTransformPoint(columnCenterWorld); startPointLocal = new Vector3(localCenter.x, localCenter.y, shelfWorkingHeight); } // 编辑器按钮手动同步列方向 public void SyncColumnDirectionFromZone() { if (assignedZone) { Vector3 colWorld = assignedZone.ColumnDirectionWorld; Debug.Log($"[{gameObject.name}] 列方向从区块同步完成"); } } void UpdateDepthDirectionFromLocal() { Vector3 localDir = depthLocalAxis switch { LocalDepthAxis.X => Vector3.right, LocalDepthAxis.Y => Vector3.up, _ => Vector3.forward }; if (!containerAttachPoint && shelf) containerAttachPoint = shelf.Find("Crane_AttachPoint") ?? shelf.Find("吸附点") ?? shelf.Find("AttachPoint"); if (shelf && shelfWorkingHeight == 0f) shelfWorkingHeight = shelf.localPosition.z; } void ValidateReferences() { if (!crane) Debug.LogError($"[{gameObject.name}] crane 未赋值！"); if (!shelf) Debug.LogError($"[{gameObject.name}] shelf 未找到！"); if (!containerAttachPoint) Debug.LogWarning($"[{gameObject.name}] containerAttachPoint 未找到！"); } void UpdateShelfColumnDirection() { Vector2 s = new Vector2(startPointLocal.x, startPointLocal.y); Vector2 e = new Vector2(endPointLocal.x, endPointLocal.y); Vector2 d = e - s; shelfColumnDir2D = d.sqrMagnitude < 1e-6f ? Vector2.right : d.normalized; } void Update() { UpdateCenterPoint(); if (isRunning) { ExecuteCraneStateMachine(); return; } if (!hasAligned && assignedZone != null) { Vehicle v = assignedZone.OccupiedVehicle; } if (hasAligned && (mode1Active || mode2Active)) { if (mode1Active) currentMode = CraneMode.Mode1; else if (mode2Active) currentMode = CraneMode.Mode2; else return; Debug.Log($"[{gameObject.name}] 对齐完成，启动模式 {currentMode}"); hasAligned = true; Debug.Log($"[{gameObject.name}] 龙门吊对齐完成"); } } bool MoveCraneToDepth(float targetDepth) { if (!crane) return false; float cur = Vector3.Dot(crane.position, worldDepthDir); float delta = targetDepth - cur; if (Mathf.Abs(delta) <= arrivalThreshold) { crane.position += delta * worldDepthDir; return true; } float spd = moveSpeed; if (Mathf.Abs(delta) < decelerationDistance && decelerationDistance > 0) spd = Mathf.Lerp(moveSpeed * 0.1f, moveSpeed, Mathf.Abs(delta) / decelerationDistance); crane.position += Mathf.Sign(delta) * Mathf.Min(spd * Time.deltaTime, Mathf.Abs(delta)) * worldDepthDir; return false; } bool MoveShelfToColumnOffset(float offset) { if (!shelf) return false; Vector2 start2D = new Vector2(startPointLocal.x, startPointLocal.y); Vector2 target2D = start2D + shelfColumnDir2D * offset; Vector2 cur2D = new Vector2(shelf.localPosition.x, shelf.localPosition.y); float dist = Vector2.Distance(cur2D, target2D); if (dist <= arrivalThreshold) { shelf.localPosition = new Vector3(target2D.x, target2D.y, shelfWorkingHeight); return true; } Vector2 n = Vector2.MoveTowards(cur2D, target2D, shelfMoveSpeed * Time.deltaTime); shelf.localPosition = new Vector3(n.x, n.y, shelfWorkingHeight); return false; } bool MoveShelfToTargetLocal(Vector3 targetLocal) { if (!shelf) return false; Vector3 tp = new Vector3(targetLocal.x, targetLocal.y, shelfWorkingHeight); Vector2 cur = new Vector2(shelf.localPosition.x, shelf.localPosition.y); Vector2 tgt = new Vector2(tp.x, tp.y); float dist = Vector2.Distance(cur, tgt); if (dist <= arrivalThreshold) { shelf.localPosition = tp; return true; } Vector2 n = Vector2.MoveTowards(cur, tgt, shelfMoveSpeed * Time.deltaTime); shelf.localPosition = new Vector3(n.x, n.y, shelfWorkingHeight); return false; } void StartLowering(float spd, float drop) { currentVerticalSpeed = spd; currentMaxDrop = drop; loweringStartZ = shelf.localPosition.z; isLowering = true; } bool ExecuteLowering() { if (!shelf) return false; float z = shelf.localPosition.z, tgt = loweringStartZ - currentMaxDrop; float nz = z - currentVerticalSpeed * Time.deltaTime; if (nz <= tgt || Mathf.Abs(z - tgt) < 0.01f) { shelf.localPosition = new Vector3(shelf.localPosition.x, shelf.localPosition.y, tgt); return true; } shelf.localPosition = new Vector3(shelf.localPosition.x, shelf.localPosition.y, nz); return false; } bool ExecuteRising() { if (!shelf) return false; float z = shelf.localPosition.z, tgt = shelfWorkingHeight; float nz = z + currentVerticalSpeed * Time.deltaTime; if (nz >= tgt || Mathf.Abs(z - tgt) < 0.01f) { shelf.localPosition = new Vector3(shelf.localPosition.x, shelf.localPosition.y, tgt); return true; } shelf.localPosition = new Vector3(shelf.localPosition.x, shelf.localPosition.y, nz); return false; } bool ExecuteRaycastDrop() { if (!containerAttachPoint) return false; Vector3 origin = containerAttachPoint.position; float dist = mode1StartRayLength; LayerMask mask = mode1StartDetectionLayers; RaycastHit[] hits = Physics.RaycastAll(origin, Vector3.down, dist, mask, QueryTriggerInteraction.Collide); foreach (var hit in hits) { if (attachedContainer != null && hit.collider.transform.IsChildOf(containerAttachPoint)) continue; Debug.Log($"[{gameObject.name}] 射线命中 {hit.collider.name}"); return true; } return false; } public GameObject DetectContainer(float radius) { if (!containerAttachPoint) return null; var hits = Physics.OverlapSphere(containerAttachPoint.position, radius); var hit = System.Array.Find(hits, c => c.CompareTag("Container") && !c.transform.IsChildOf(shelf)); return hit ? hit.gameObject : null; } public GameObject DetectCargoOnVehicle(float radius) { if (!containerAttachPoint) return null; var hits = Physics.OverlapSphere(containerAttachPoint.position, radius); var hit = System.Array.Find(hits, c => c.CompareTag("Container") && c.GetComponentInParent() != null); return hit ? hit.gameObject : null; } public Vehicle DetectVehicleComponent(float radius) { if (!containerAttachPoint) return null; var hits = Physics.OverlapSphere(containerAttachPoint.position, radius); var hit = System.Array.Find(hits, c => c.GetComponent() || c.GetComponentInParent()); return hit ? (hit.GetComponent() ?? hit.GetComponentInParent()) : null; } void AttachContainerOnly(GameObject o) { if (!o || !containerAttachPoint) return; SetKinematic(o, true); o.transform.SetParent(containerAttachPoint, true); attachedContainer = o; } void SetKinematic(GameObject o, bool k) { var rb = o?.GetComponent(); if (rb) { rb.isKinematic = k; rb.useGravity = !k; rb.velocity = rb.angularVelocity = Vector3.zero; } } void RequestVehicleToLeave() { if (currentVehicle != null) currentVehicle.occupiedBlock?.ReleaseVehicle(currentVehicle); } void ExecuteCraneStateMachine() { switch (currentState) { case CraneState.MovingToEnd: if (MoveShelfToTargetLocal(endPointLocal)) { currentState = CraneState.LoweringAtEnd; StartLowering(endPointVerticalSpeed, endPointMaxDrop); } break; case CraneState.LoweringAtEnd: { var cargo = DetectCargoOnVehicle(mode1EndVehicleRadius); if (cargo) { AttachContainerOnly(cargo); hasPickedUpCargo = true; pickupRetryCount = 0; Debug.Log($"[{gameObject.name}] 从货车取货成功: {cargo.name}"); RiseAfterLower(); return; } if (ExecuteLowering()) { cargo = DetectCargoOnVehicle(mode1EndVehicleRadius); if (cargo) { AttachContainerOnly(cargo); hasPickedUpCargo = true; pickupRetryCount = 0; RiseAfterLower(); } else HandlePickupFailure(); } } break; case CraneState.Rising: if (ExecuteRising()) { if (retryPickupFlag) { retryPickupFlag = false; isRunning = false; hasAligned = false; isMoving = false; currentState = CraneState.Idle; } else if (hasDroppedCargo) { RequestVehicleToLeave(); Debug.Log($"[{gameObject.name}] 作业完成，货车离开"); Stop(); } else if (hasPickedUpCargo) { if (currentMode == CraneMode.Mode1) { if (assignedZone && assignedZone.GetNextMatrixPosition(out targetCraneDepth, out targetShelfColumnOffset, out targetShelfHeightDrop, out currentHeightLevel)) currentState = CraneState.MovingCraneToDepth; else Stop(); } else if (currentMode == CraneMode.Mode2) { if (currentTarget != null) { CalculateTargetDepth(); currentState = CraneState.MovingCraneToVehicle; } else Stop(); } } } break; case CraneState.MovingCraneToDepth: if (MoveCraneToDepth(targetCraneDepth)) currentState = CraneState.MovingShelfToColumn; break; case CraneState.MovingShelfToColumn: if (MoveShelfToColumnOffset(targetShelfColumnOffset)) { if (shelf) shelf.localPosition = new Vector3(shelf.localPosition.x, shelf.localPosition.y, shelfWorkingHeight); currentState = CraneState.RaycastDropping; StartLowering(startPointVerticalSpeed, startPointMaxDrop + targetShelfHeightDrop); } break; case CraneState.RaycastDropping: if (ExecuteRaycastDrop() && attachedContainer) { ReleaseContainerToZone(); hasDroppedCargo = true; Debug.Log($"[{gameObject.name}] 放货至矩阵完成"); RiseAfterLower(); return; } if (ExecuteLowering()) { if (attachedContainer) { ReleaseContainerToZone(); hasDroppedCargo = true; } RiseAfterLower(); } break; case CraneState.MovingCraneToDepthForPickup: if (assignedZone && assignedZone.GetLastMatrixPosition(out targetCraneDepth, out targetShelfColumnOffset, out targetShelfHeightDrop, out currentHeightLevel)) { if (MoveCraneToDepth(targetCraneDepth)) currentState = CraneState.MovingShelfToColumnForPickup; } else Stop(); break; case CraneState.MovingShelfToColumnForPickup: if (MoveShelfToColumnOffset(targetShelfColumnOffset)) { if (shelf) shelf.localPosition = new Vector3(shelf.localPosition.x, shelf.localPosition.y, shelfWorkingHeight); currentState = CraneState.LoweringAtStart; StartLowering(startPointVerticalSpeed, startPointMaxDrop + targetShelfHeightDrop); } break; case CraneState.LoweringAtStart: { var cont = DetectContainer(mode2StartCargoRadius); if (cont) { AttachContainerOnly(cont); hasPickedUpCargo = true; pickupRetryCount = 0; assignedZone?.DecrementCount(); RiseAfterLower(); } else HandlePickupFailure(); } } break; case CraneState.MovingCraneToVehicle: if (currentTarget) { CalculateTargetDepth(); if (MoveCraneToDepth(targetVehicleDepth)) currentState = CraneState.MovingShelfToEnd; } else Stop(); break; case CraneState.MovingShelfToEnd: if (MoveShelfToTargetLocal(endPointLocal)) { currentState = CraneState.LoweringAtEndForDrop; RiseAfterLower(); } } break; case CraneState.Completed: break; } } void ReleaseContainerToZone() { if (!attachedContainer) return; if (assignedZone != null) { int targetIndex = assignedZone.PlacedCount; Vector3 targetWorldPos = assignedZone.GetWorldPosition(targetIndex); attachedContainer.transform.position = targetWorldPos; attachedContainer.transform.rotation = Quaternion.identity; Debug.Log($"[{gameObject.name}] 精确放置到矩阵位置[{targetIndex}]: {targetWorldPos}"); } SetKinematic(attachedContainer, false); attachedContainer.transform.SetParent(null); assignedZone?.IncrementCount(); attachedContainer = null; } void TransferToVehicle(Vehicle vehicle) { Transform a = vehicle.transform.Find("CargoAttachPoint") ?? vehicle.transform; attachedContainer.transform.SetParent(null); attachedContainer.transform.SetParent(a); SetKinematic(attachedContainer, false); Debug.Log($"[{gameObject.name}] 货物移交货车: {attachedContainer.name}"); attachedContainer = null; } void HandlePickupFailure() { if (pickupRetryCount < maxPickupRetries) { pickupRetryCount++; Debug.Log($"[{gameObject.name}] 取货失败，重试 {pickupRetryCount}/{maxPickupRetries}"); retryPickupFlag = true; RiseAfterLower(); } else { Debug.Log($"[{gameObject.name}] 重试耗尽，释放车辆"); RequestVehicleToLeave(); Stop(); } } void RiseAfterLower() { currentState = CraneState.Rising; isLowering = false; } void StartMode(CraneMode m) { currentMode = m; Debug.Log($"[{gameObject.name}] Mode1 启动"); } else if (m == CraneMode.Mode2) { currentState = CraneState.MovingCraneToDepthForPickup; Debug.Log($"[{gameObject.name}] Mode2 启动"); } } public void Stop() { isRunning = false; } else cachedCenter = transform.position + centerOffset; } // 编辑器按钮方法 public void SetStartPoint() { if (shelf) { startPointLocal = shelf.localPosition; startPointLocal.z = shelfWorkingHeight; SyncColumnDirectionIfNeeded();
